- hooks/discord-notify.sh adds an exponential-backoff retry loop around the existing curl POST. Recommended shape:
  ```
  attempts=0
  max_attempts=${DISCORD_NOTIFY_MAX_ATTEMPTS:-3}
  while [ $attempts -lt $max_attempts ]; do
    if curl -fsS -X POST -H 'Content-Type: application/json' \
      --max-time "${DISCORD_NOTIFY_TIMEOUT:-10}" \
      -d "$payload" "$DISCORD_NOTIFY_WEBHOOK" >/dev/null 2>&1; then
      ok=1; break
    fi
    sleep_for=$(( 2 ** attempts ))
    sleep $sleep_for
    attempts=$((attempts + 1))
  done
  if [ "${ok:-0}" -ne 1 ]; then
    # dead-letter
    printf '%s payload=%s\n' "$timestamp" "$payload" >> "$STATE_DIR/discord-notify-deadletter.log"
  fi
  ```
